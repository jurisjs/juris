Code Examples

Here's how these components are created:

// Simple Counter Component
const counterComponent = (props, context) => ({
    div: {
        style: { 
            background: props.theme || 'linear-gradient(135deg, #667eea, #764ba2)',
            padding: '20px', 
            borderRadius: '8px', 
            color: 'white', 
            textAlign: 'center' 
        },
        children: [
            { h3: { text: `Counter: ${context.component.getState('count', 0)}` } },
            {
                button: {
                    text: '+ Increment',
                    style: { margin: '0 5px', padding: '8px 16px' },
                    onclick: () => {
                        const current = context.component.getState('count', 0);
                        context.component.setState('count', current + 1);
                    }
                }
            }
        ]
    }
});

// Register and create the component
juris.webComponentFactory.create('simple-counter', counterComponent, {
    attributes: ['theme', 'initial-count'],
    styles: ':host { display: block; margin: 10px 0; }'
});
